local MINECRAFTIA_FACE = nil
do
    local hasAPI = (writefile and isfile and getcustomasset and HttpService)
    if hasAPI then
        pcall(function()
            local fileName = "Minecraftia-Regular.ttf"
            local fontJson = "Minecraftia.font"
            if not isfile(fileName) then
                writefile(fileName, game:HttpGet("https://github.com/i77lhm/storage/raw/refs/heads/main/fonts/Minecraftia-Regular.ttf"))
            end
            if isfile(fontJson) then delfile(fontJson) end
            local data = { name = "Minecraftia", faces = {{ name = "Regular", weight = 400, style = "Normal", assetId = getcustomasset(fileName) }}}
            writefile(fontJson, HttpService:JSONEncode(data))
            MINECRAFTIA_FACE = Font.new(getcustomasset(fontJson), Enum.FontWeight.Regular, Enum.FontStyle.Normal)
        end)
    end
end

local function ApplyFont(label)
    if MINECRAFTIA_FACE then
        pcall(function() label.FontFace = MINECRAFTIA_FACE end)
    else
        label.Font = Enum.Font.GothamBold
    end
end

local function GetWeaponIcon(weaponName)
    if not weaponName then return nil end
    local lower = string.lower(weaponName)
    for _, key in ipairs(WeaponIconOrder) do
        if string.find(lower, key, 1, true) then return WeaponIcons[key] end
    end
    if WeaponIcons[lower] then return WeaponIcons[lower] end
    return nil
end

local function FindEquippedTool(char, player)
    if not char then return nil end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then return tool end
    for _, c in ipairs(char:GetChildren()) do
        if c:IsA("Tool") then return c end
    end
    for _, c in ipairs(char:GetChildren()) do
        if (c:IsA("Model") or c:IsA("Folder")) and c:FindFirstChild("Handle") then
            return c
        end
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        for _, c in ipairs(hum:GetChildren()) do
            if c:IsA("Tool") then return c end
        end
    end
    return nil
end

local function BoxSolve(torso)
    if not torso then return nil end
    local size = torso.Size
    local top = torso.Position + v3(0, size.Y / 2 + 0.6, 0)
    local bottom = torso.Position - v3(0, size.Y / 2 + 2.0, 0)
    local dist = (torso.Position - Workspace.CurrentCamera.CFrame.Position).Magnitude
    local topSP, topOn = Workspace.CurrentCamera:WorldToViewportPoint(top)
    local botSP, botOn = Workspace.CurrentCamera:WorldToViewportPoint(bottom)
    local onScreen = topOn and botOn
    local h = math.max(mAbs(botSP.Y - topSP.Y), 3)
    local w = h / 1.8
    local center = v2((topSP.X + botSP.X) * 0.5, (topSP.Y + botSP.Y) * 0.5)
    return v2(w, h), center, onScreen, mFloor(dist)
end

local function UpdateESP()
    local cam = Workspace.CurrentCamera
    if not cam then return end

    for player, Data in pairs(EspPlayers) do
        if not Data or not Data.Root then continue end
        local hide = not Settings.Enabled
            or (player == localPlayer and not Settings.SelfESPEnabled)
        if hide then
            Data.Root.Visible = false
            continue
        end
        local char = player.Character
        if not char then
            Data.Root.Visible = false
            continue
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = hum and hum.RootPart
        if not root or (hum and hum.Health <= 0) then
            Data.Root.Visible = false
            continue
        end

        local size, center, onScreen, dist
        if Settings.ThirdPerson.Enabled then
            local topPos = root.Position + v3(0, 2.5, 0)
            local bottomPos = root.Position - v3(0, 3.0, 0)
            local topSP, topOn = cam:WorldToViewportPoint(topPos)
            local botSP, botOn = cam:WorldToViewportPoint(bottomPos)
            onScreen = topOn and botOn
            if onScreen then
                local h = math.max(mAbs(botSP.Y - topSP.Y), 3)
                local w = h / 1.8
                size = v2(w, h)
                center = v2((topSP.X + botSP.X) * 0.5, (topSP.Y + botSP.Y) * 0.5)
                dist = mFloor((root.Position - cam.CFrame.Position).Magnitude)
            end
        else
            size, center, onScreen, dist = BoxSolve(root)
        end

        if not size or not onScreen or dist > Settings.RenderDistance then
            Data.Root.Visible = false
            continue
        end

        local boxColor = Settings.BoxColor

        Data.Root.Visible = true
        Data.Root.Position = UDim2.fromOffset(center.X, center.Y)
        Data.Root.Size = UDim2.fromOffset(size.X, size.Y)

        if Data.BoxFill then
            if Settings.BoxFillEnabled and Settings.BoxEnabled then
                Data.BoxFill.Visible = true
                Data.BoxFill.BackgroundTransparency = Settings.BoxFillTransp
                Data.BoxFillGradient.Color = csNew({
                    cskNew(0, Settings.BoxFillColor1),
                    cskNew(1, Settings.BoxFillColor2),
                })
                Data.BoxFillGradient.Rotation = Settings.BoxFillRotation
            else
                Data.BoxFill.Visible = false
            end
        end
        if Settings.BoxEnabled then
            Data.BoxOuterFrame.Visible = true
            Data.BoxInnerFrame.Visible = true
            Data.BoxMainFrame.Visible = true
            Data.OuterStroke.Color = Settings.BoxOutlineColor
            Data.OuterStroke.Thickness = Settings.BoxOutlineThick
            Data.InnerStroke.Color = Settings.BoxOutlineColor
            Data.InnerStroke.Thickness = Settings.BoxOutlineThick
            Data.MainStroke.Color = boxColor
            Data.MainStroke.Thickness = Settings.BoxThickness
        else
            Data.BoxOuterFrame.Visible = false
            Data.BoxInnerFrame.Visible = false
            Data.BoxMainFrame.Visible = false
        end
        if Data.HPOutline and Settings.HealthBar and hum then
            local pct = mClamp(hum.Health / hum.MaxHealth, 0, 1)
            Data.HPOutline.Visible = true
            Data.HPOutline.Size = UDim2.fromOffset(6, size.Y + 2)
            Data.HPOutline.Position = UDim2.new(0, -4, 0, -1)
            Data.HPFill.Size = UDim2.new(1, 0, pct, 0)
            Data.HPGrad.Color = csNew({
                cskNew(0.0, Settings.HealthBarColor1),
                cskNew(0.5, Settings.HealthBarColor2),
                cskNew(1.0, Settings.HealthBarColor1),
            })
        elseif Data.HPOutline then
            Data.HPOutline.Visible = false
        end
        if Data.Name then
            Data.Name.Visible = Settings.NameEnabled
            if Settings.NameEnabled then
                Data.Name.Text = player.Name
                Data.Name.TextColor3 = Settings.NameColor
                Data.Name.TextSize = Settings.NameSize
            end
        end
        if Data.Distance then
            Data.Distance.Visible = Settings.DistEnabled
            if Settings.DistEnabled then
                Data.Distance.Text = "[" .. dist .. "]"
                Data.Distance.TextColor3 = Settings.DistColor
                Data.Distance.TextSize = Settings.DistSize
            end
        end
        if Data.Weapon then
            if Settings.WeaponEnabled then
                local tool = FindEquippedTool(char, player)
                Data.Weapon.Visible = true
                Data.Weapon.TextColor3 = Settings.WeaponColor
                Data.Weapon.TextSize = Settings.WeaponSize
                if tool then
                    Data.Weapon.Text = "[" .. tool.Name .. "]"
                    if Data.WeaponIcon then
                        local icon = GetWeaponIcon(tool.Name)
                        if icon then
                            Data.WeaponIcon.Image = icon
                            Data.WeaponIcon.Visible = true
                        else
                            Data.WeaponIcon.Visible = false
                        end
                    end
                else
                    Data.Weapon.Text = "[None]"
                    if Data.WeaponIcon then Data.WeaponIcon.Visible = false end
                end
            else
                Data.Weapon.Visible = false
                if Data.WeaponIcon then Data.WeaponIcon.Visible = false end
            end
        end
    end
end

do
    local ignoreCache = {}
    local lastIgnore = 0

    local function getFovOrigin()
        local cam = Workspace.CurrentCamera
        if not cam then return v2(0, 0) end
        local vp = cam.ViewportSize
        return v2(vp.X * 0.5, vp.Y * 0.5)
    end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.IgnoreWater = true

    local function getIgnoreList()
        local now = tick()
        if now - lastIgnore < 0.5 then return ignoreCache end
        lastIgnore = now
        ignoreCache = { localPlayer.Character, Workspace.CurrentCamera }
        for plr in next, F.playerCache do
            if plr.Character then ignoreCache[#ignoreCache + 1] = plr.Character end
        end
        return ignoreCache
    end

    local function isInsideBuildings(inst)
        local node = inst
        while node and node ~= Workspace do
            if node.Name == "Buildings" and node.Parent == Workspace then return true end
            node = node.Parent
        end
        return false
    end

    function F.isTeammate() return false end

    function F.pointVisible(character, worldPoint)
        local cam = Workspace.CurrentCamera
        if not cam then return false end

        local origin = cam.CFrame.Position
        local ignored = table.clone(getIgnoreList())

        for _ = 1, 12 do
            rayParams.FilterDescendantsInstances = ignored
            local direction = worldPoint - origin
            if direction.Magnitude <= 0.01 then return true end

            local result = Workspace:Raycast(origin, direction, rayParams)
            if not result then return true end

            local inst = result.Instance
            if inst and inst:IsDescendantOf(character) then return true end

            if inst and inst:IsA("BasePart") and not inst.CanCollide and not isInsideBuildings(inst) then
                ignored[#ignored + 1] = inst
            else
                return false
            end
        end
        return false
    end

    function F.getClosestPlayerPW()
        local closestPart, closestPos, bestScore = nil, nil, math.huge
        local fovOrigin = getFovOrigin()
        local camPos = Workspace.CurrentCamera.CFrame.Position
        local myRoot = F.char and F.char:FindFirstChild("HumanoidRootPart")
        local myPos = myRoot and myRoot.Position or camPos

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= localPlayer then
                local char = player.Character
                if char then
                    local hum = char:FindFirstChild("Humanoid")
                    if hum and hum.Health > 0 then
                        local part = char:FindFirstChild(Settings.SA.TargetPart) or char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
                        if part then
                            local worldDist = (part.Position - myPos).Magnitude
                            if worldDist <= Settings.SA.MaxDistance then
                                local targetPos = part.Position
                                local cam = Workspace.CurrentCamera
                                local sp, onScreen = cam:WorldToViewportPoint(targetPos)
                                if onScreen then
                                    local fovDist = (fovOrigin - v2(sp.X, sp.Y)).Magnitude
                                    if fovDist <= Settings.SA.FOVRadius then
                                        local score = Settings.SA.TargetMode == "Cursor" and fovDist or worldDist
                                        if score < bestScore then
                                            bestScore = score
                                            closestPart = part
                                            closestPos = targetPos
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
        return closestPart, closestPos
    end
end

do
    local shots = {}

    function F.recordShot(origin, dirUnit)
        shots[#shots + 1] = { time = osClock(), origin = origin, dir = dirUnit }
        if #shots > 25 then table.remove(shots, 1) end
    end

    function F.markShotTarget()
        local part = F.targetPart
        local plr = part and part.Parent and Players:GetPlayerFromCharacter(part.Parent)
        if plr then
            F.lastConfirmedShotPlayer = plr
            F.lastConfirmedShotTime = osClock()
        end
    end

    function F.isOnOurShot(rootPos)
        local now = osClock()
        for i = #shots, 1, -1 do
            local shot = shots[i]
            if now - shot.time > 6 then
                table.remove(shots, i)
            else
                local toTarget = rootPos - shot.origin
                local along = toTarget:Dot(shot.dir)
                if along > 0 then
                    local perp = (toTarget - shot.dir * along).Magnitude
                    local tolerance = mClamp(2.25 + along * 0.002, 2.25, 8)
                    if perp <= tolerance then return true end
                end
            end
        end
        return false
    end
end

do
    local pool, poolSize = {}, 0
    local active = {}

    local function getTr()
        if poolSize > 0 then
            local p = pool[poolSize]
            pool[poolSize] = nil
            poolSize = poolSize - 1
            return p
        end
        local p = Instance.new("Part")
        p.Anchored = true
        p.CanCollide = false
        p.Transparency = 1
        p.Size = v3(0.1, 0.1, 0.1)
        local a0 = Instance.new("Attachment", p)
        a0.Name = "A0"
        local a1 = Instance.new("Attachment", p)
        a1.Name = "A1"
        local b = Instance.new("Beam")
        b.Name = "B"
        b.Attachment0 = a0
        b.Attachment1 = a1
        b.FaceCamera = true
        b.LightEmission = 1
        b.LightInfluence = 0
        b.TextureLength = 2
        b.TextureSpeed = 2
        b.Parent = p
        return p
    end

    local function release(p, b)
        if b then b.Enabled = false end
        if p then
            p.Parent = nil
            poolSize = poolSize + 1
            pool[poolSize] = p
        end
    end

    F.createTrace = function(startPos, endPos)
        if F.Unloaded or not Settings.Tracers.Enabled then return end
        local cam = Workspace.CurrentCamera
        if cam then
            local d = (startPos - cam.CFrame.Position).Magnitude
            if d < 4 then
                startPos = (cam.CFrame * cNew(2.5, -1.5, -1)).Position
            end
        end
        local p = getTr()
        p.Parent = Workspace
        local a0 = p:FindFirstChild("A0")
        local a1 = p:FindFirstChild("A1")
        if a0 then a0.WorldPosition = startPos end
        if a1 then a1.WorldPosition = endPos end
        local b = p:FindFirstChild("B")
        if not b then release(p) return end
        b.Color = ColorSequence.new(Settings.Tracers.Color)
        b.Transparency = NumberSequence.new(Settings.Tracers.Transparency)
        b.Texture = Settings.Tracers.TextureID
        b.Width0 = Settings.Tracers.Width
        b.Width1 = Settings.Tracers.Width
        b.Enabled = true
        active[#active + 1] = { p = p, b = b, start = tick() }
    end

    F.spawnTracerFromShot = function(origin, direction)
        if not Settings.Tracers.Enabled then return end
        local mag = direction.Magnitude
        local dir = direction.Unit
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        rayParams.FilterDescendantsInstances = { localPlayer.Character, Workspace.CurrentCamera }
        local result = Workspace:Raycast(origin, dir * (mag > 0 and mag or 300), rayParams)
        local endPos = result and result.Position or (origin + dir * 300)
        F.createTrace(origin, endPos)
    end

    F.updateTracers = function()
        if #active == 0 then return end
        local lt = Settings.Tracers.LifeTime
        for i = #active, 1, -1 do
            local t = active[i]
            local p, b = t.p, t.b
            if not p or not p.Parent then
                table.remove(active, i)
            else
                local alpha = 1 - ((tick() - t.start) / lt)
                if alpha <= 0 then
                    release(p, b)
                    table.remove(active, i)
                elseif b then
                    b.Transparency = NumberSequence.new(alpha * Settings.Tracers.Transparency)
                end
            end
        end
    end

    F.destroyAllTracers = function()
        for i = #active, 1, -1 do
            local t = active[i]
            if t and t.p then pcall(function() t.p:Destroy() end) end
            active[i] = nil
        end
        for i = 1, poolSize do pcall(function() pool[i]:Destroy() end) end
        pool = {}
        poolSize = 0
    end
end

do
    local backtrackGhost = nil
    local backtrackParts = {}
    local lastSnap = 0
    local backtrackFolder = nil

    local function cleanupBacktrack()
        if backtrackGhost then
            pcall(function() backtrackGhost:Destroy() end)
            backtrackGhost = nil
        end
        backtrackParts = {}
        if backtrackFolder then
            pcall(function() backtrackFolder:Destroy() end)
            backtrackFolder = nil
        end
    end

    local function hardenPart(inst)
        inst.Anchored = true
        inst.CanCollide = false
        inst.CanQuery = false
        inst.CanTouch = false
        inst.Massless = true
        inst.CastShadow = false
        inst.Material = Enum.Material.ForceField
        inst.Color = Settings.Backtrack.Color
        inst.Transparency = Settings.Backtrack.Transparency
        inst.CustomPhysicalProperties = PhysicalProperties.new(0.01, 0.01, 0.01, 0, 0)
        inst.AssemblyLinearVelocity = v3(0, 0, 0)
        inst.AssemblyAngularVelocity = v3(0, 0, 0)
        pcall(function()
            inst.CollisionGroup = "FloppaBacktrack"
        end)
    end

    local function buildGhost(char)
        cleanupBacktrack()
        if not char then return end

        local wasArch = char.Archivable
        char.Archivable = true
        local clone
        pcall(function() clone = char:Clone() end)
        char.Archivable = wasArch
        if not clone then return end

        local toDestroy = {}
        for _, inst in ipairs(clone:GetDescendants()) do
            if inst:IsA("BasePart") then
                hardenPart(inst)
                backtrackParts[inst.Name] = inst
            elseif inst:IsA("Decal") or inst:IsA("Texture") then
                inst.Transparency = 1
            elseif inst:IsA("Script") or inst:IsA("LocalScript") or inst:IsA("ModuleScript")
                or inst:IsA("Accessory") or inst:IsA("Motor6D") or inst:IsA("Motor")
                or inst:IsA("Weld") or inst:IsA("WeldConstraint") or inst:IsA("JointInstance")
                or inst:IsA("Attachment") or inst:IsA("Trail") or inst:IsA("ParticleEmitter")
                or inst:IsA("Sound") or inst:IsA("Highlight") or inst:IsA("PointLight")
                or inst:IsA("SpotLight") or inst:IsA("SurfaceLight") or inst:IsA("Sparkles")
                or inst:IsA("Fire") or inst:IsA("Smoke") or inst:IsA("Explosion")
                or inst:IsA("ForceField") or inst:IsA("SelectionBox")
                or inst:IsA("SurfaceAppearance") then
                toDestroy[#toDestroy + 1] = inst
            end
        end
        for _, inst in ipairs(toDestroy) do
            pcall(function() inst:Destroy() end)
        end

        for _, inst in ipairs(clone:GetDescendants()) do
            if inst:IsA("BasePart") then hardenPart(inst) end
        end

        clone.Name = "FloppaBacktrack"
        pcall(function() clone:SetAttribute("FloppaBacktrack", true) end)
        clone.Parent = Workspace
        backtrackGhost = clone
    end

    local function updateGhostTransforms(char)
        if not backtrackGhost then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end

        local backDir = root.CFrame.LookVector * -1
        local offset = backDir * (Settings.Backtrack.OffsetBack or 3)

        for name, ghostPart in pairs(backtrackParts) do
            local realPart = char:FindFirstChild(name)
            if realPart and realPart:IsA("BasePart") then
                ghostPart.CFrame = realPart.CFrame + offset
            end
        end
    end

    F.updateBacktrack = function(dt)
        if not Settings.Backtrack.Enabled then
            if backtrackGhost then cleanupBacktrack() end
            return
        end

        local char = F.char
        if not char then
            if backtrackGhost then cleanupBacktrack() end
            return
        end

        if not backtrackGhost or not backtrackGhost.Parent then
            buildGhost(char)
            return
        end

        updateGhostTransforms(char)

        local now = osClock()
        if now - lastSnap >= Settings.Backtrack.UpdateRate then
            lastSnap = now
        end
    end

    F.cleanupBacktrack = cleanupBacktrack
end

do
    if hookmetamethod and getnamecallmethod and newcclosure then
        local old
        old = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            local method = getnamecallmethod()

            if method == "FireServer" and Settings.SA.Enabled and F.targetPos then
                if math.random(1, 100) <= (Settings.SA.HitChance or 100) then
                    local args = table.pack(...)
                    local origin
                    for i = 1, args.n do
                        local a = args[i]
                        if typeof(a) == "Vector3" then
                            origin = origin or a
                            args[i] = F.targetPos
                        elseif typeof(a) == "CFrame" then
                            origin = origin or a.Position
                            args[i] = cNew(F.targetPos)
                        end
                    end
                    if origin and F.targetPos and Settings.Tracers.Enabled then
                        task.spawn(F.spawnTracerFromShot, origin, F.targetPos - origin)
                    end
                    return old(self, tUnpack(args, 1, args.n))
                end
            elseif method == "Raycast" and self == Workspace then
                if checkcaller() then return old(self, ...) end
                local origin, direction, rayParams = ...
                if typeof(origin) ~= "Vector3" or typeof(direction) ~= "Vector3" then
                    return old(self, ...)
                end
                local mag = direction.Magnitude
                if mag <= 50 then return old(self, ...) end

                local camPos = Workspace.CurrentCamera.CFrame.Position
                if (origin - camPos).Magnitude <= 5 then
                    F.recordShot(origin, direction.Unit)
                end

                local newOrigin, newDir = origin, direction
                if Settings.SA.Enabled and F.targetPos then
                    newDir = (F.targetPos - newOrigin).Unit * mag
                end

                return old(self, newOrigin, newDir, rayParams)
            end

            return old(self, ...)
        end))
    end
end

                            buffer.writef32(v, 8, pos.Y)
                            buffer.writef32(v, 12, pos.Z)
                        end)
                        patched = true
                    end
                end
                if patched then
                    F.markShotTarget()
                    if origin and Settings.Tracers.Enabled then
                        task.spawn(F.spawnTracerFromShot, origin, pos - origin)
                    end
                    return origFire(self, tUnpack(args, 1, args.n))
                end
            end
            return origFire(self, ...)
        end
    end
end

do
    local ok, GunBase = pcall(function()
        return require(RS:WaitForChild("Gun"):WaitForChild("Scripts"):WaitForChild("GunBase"))
    end)

    if ok and type(GunBase) == "table" then
        local fireName = type(GunBase.fire) == "function" and "fire"
            or type(GunBase.Fire) == "function" and "Fire"
        local reloadName
        for _, candidate in ipairs({ "reload", "Reload", "startReload", "beginReload" }) do
            if type(GunBase[candidate]) == "function" then
                reloadName = candidate
                break
            end
        end

        if fireName and reloadName then
            local originalFire = GunBase[fireName]
            local originalReload = GunBase[reloadName]
            local wrappedFire, wrappedReload
            local activeGun, ammoGun = nil, nil
            local emptyHandled = false
            local aiming = false
            local interruptedByAim = false
            local syntheticUntil = 0
            local shotRecoveryUntil = 0
            local reloadRetryAt = 0
            local lastCheck = 0
            local MAG_KEYS = {
                "Ammo", "ammo", "CurrentAmmo", "currentAmmo", "current_ammo",
                "AmmoInMag", "ammoInMag", "Magazine", "magazine", "Mag", "mag",
                "Clip", "clip", "Bullets", "bullets", "Rounds", "rounds",
            }
            local CONTAINERS = { "State", "state", "Data", "data", "Stats", "stats", "Weapon", "weapon", "Gun", "gun" }

            local function numberValue(value)
                if type(value) == "number" then return value end
                if typeof(value) == "Instance" and (value:IsA("IntValue") or value:IsA("NumberValue")) then
                    return value.Value
                end
                if type(value) == "string" then return tonumber(value) end
                return nil
            end

            local function readDirect(source)
                if type(source) == "table" then
                    for _, key in ipairs(MAG_KEYS) do
                        local value = numberValue(rawget(source, key))
                        if value ~= nil then return value end
                    end
                elseif typeof(source) == "Instance" then
                    for _, key in ipairs(MAG_KEYS) do
                        local value = numberValue(source:GetAttribute(key)) or numberValue(source:FindFirstChild(key))
                        if value ~= nil then return value end
                    end
                end
            end

            local function readAmmo(source)
                local value = readDirect(source)
                if value ~= nil then return value end
                if type(source) == "table" then
                    for _, key in ipairs(CONTAINERS) do
                        local child = rawget(source, key)
                        if child and child ~= source then
                            value = readDirect(child)
                            if value ~= nil then return value end
                        end
                    end
                elseif typeof(source) == "Instance" then
                    for _, key in ipairs(CONTAINERS) do
                        local child = source:FindFirstChild(key)
                        if child then
                            value = readDirect(child)
                            if value ~= nil then return value end
                        end
                    end
                end
            end

            local function rememberGun(self)
                if not self then return end
                activeGun = self
                if readAmmo(self) ~= nil then ammoGun = self end
            end

            local function setReloading(value, duration)
                Settings.Reload.IsReloading = value
                Settings.Reload.ReloadUntil = value and (osClock() + (duration or 3.2)) or 0
            end

            local function sendReloadKey()
                syntheticUntil = osClock() + 0.75
                local env = getgenv()
                local press, release = rawget(env, "keypress"), rawget(env, "keyrelease")
                if type(press) == "function" and type(release) == "function" then
                    local sent = pcall(press, 0x52)
                    task.delay(0.05, function() pcall(release, 0x52) end)
                    if sent then return true end
                end
                local okVim, vim = pcall(game.GetService, game, "VirtualInputManager")
                if okVim and vim then
                    local sent = pcall(function()
                        vim:SendKeyEvent(true, Enum.KeyCode.R, false, game)
                        task.delay(0.05, function()
                            pcall(function() vim:SendKeyEvent(false, Enum.KeyCode.R, false, game) end)
                        end)
                    end)
                    if sent then return true end
                end
                syntheticUntil = 0
                return false
            end

            local function actionAnimationPlaying()
                local character = F.char or localPlayer.Character
                local humanoid = character and character:FindFirstChildWhichIsA("Humanoid")
                if not humanoid then return false end
                local ok, tracks = pcall(function() return humanoid:GetPlayingAnimationTracks() end)
                if not ok or type(tracks) ~= "table" then return false end
                for _, track in ipairs(tracks) do
                    if track.IsPlaying then
                        local name = sLower(tostring(track.Name or ""))
                        if sFind(name, "fire", 1, true)
                            or sFind(name, "shoot", 1, true)
                            or sFind(name, "shot", 1, true)
                            or sFind(name, "bolt", 1, true)
                            or sFind(name, "pump", 1, true)
                            or sFind(name, "chamber", 1, true)
                            or sFind(name, "cycle", 1, true) then
                            return true
                        end
                    end
                end
                return false
            end

            local function tryAutoReload()
                local now = osClock()
                if F.Unloaded or not Settings.Reload.Auto or not activeGun or emptyHandled then return end
                if now < shotRecoveryUntil or now < reloadRetryAt then return end
                local ammo = readAmmo(ammoGun or activeGun)
                if ammo ~= 0 then return end
                if actionAnimationPlaying() then
                    reloadRetryAt = now + 0.08
                    return
                end
                emptyHandled = true
                setReloading(true, 3.2)
                if not sendReloadKey() then
                    local okCall, result = pcall(originalReload, activeGun)
                    if not okCall or result == false then
                        emptyHandled = false
                        setReloading(false)
                    end
                end
            end

            wrappedReload = function(self, ...)
                rememberGun(self)
                local synthetic = osClock() <= syntheticUntil
                local results = table.pack(originalReload(self, ...))
                if results[1] ~= false then
                    emptyHandled = true
                    setReloading(true, tonumber(rawget(self, "ReloadTime") or rawget(self, "ReloadDuration")) or 3.2)
                elseif not synthetic and Settings.Reload.Auto then
                    task.defer(function()
                        emptyHandled = false
                        tryAutoReload()
                    end)
                end
                return tUnpack(results, 1, results.n)
            end

            wrappedFire = function(self, ...)
                rememberGun(self)
                emptyHandled = false
                local results = table.pack(originalFire(self, ...))
                local recovery = numberValue(rawget(self, "FireDelay"))
                    or numberValue(rawget(self, "FireCooldown"))
                    or numberValue(rawget(self, "ShotDelay"))
                    or 0.12
                shotRecoveryUntil = osClock() + math.max(recovery, 0.12)
                task.defer(tryAutoReload)
                return tUnpack(results, 1, results.n)
            end

            GunBase[reloadName] = wrappedReload
            GunBase[fireName] = wrappedFire

            F.isWeaponReloading = function()
                local ammo = readAmmo(ammoGun or activeGun)
                if ammo and ammo > 0 and not Settings.Reload.IsReloading then
                    emptyHandled = false
                    interruptedByAim = false
                    setReloading(false)
                elseif Settings.Reload.IsReloading and osClock() >= Settings.Reload.ReloadUntil and ammo ~= 0 then
                    setReloading(false)
                end
                return Settings.Reload.IsReloading
            end

            F.trackConn(UIS.InputBegan:Connect(function(input)
                if input.UserInputType ~= Enum.UserInputType.MouseButton2 then return end
                aiming = true
                if Settings.Reload.IsReloading and readAmmo(ammoGun or activeGun) == 0 then
                    interruptedByAim = true
                end
            end))

            F.trackConn(UIS.InputEnded:Connect(function(input)
                if input.UserInputType ~= Enum.UserInputType.MouseButton2 then return end
                aiming = false
                if interruptedByAim and readAmmo(ammoGun or activeGun) == 0 then
                    interruptedByAim = false
                    emptyHandled = false
                    setReloading(false)
                    task.defer(tryAutoReload)
                end
            end))

            F.trackConn(RunService.Heartbeat:Connect(function()
                if F.Unloaded then return end
                local now = osClock()
                if now - lastCheck < 0.05 then return end
                lastCheck = now
                local ammo = readAmmo(ammoGun or activeGun)
                if ammo and ammo > 0 then
                    emptyHandled = false
                    interruptedByAim = false
                    setReloading(false)
                elseif ammo == 0 then
                    if interruptedByAim and not aiming then
                        interruptedByAim = false
                        emptyHandled = false
                        setReloading(false)
                    end
                    tryAutoReload()
                elseif Settings.Reload.IsReloading and now >= Settings.Reload.ReloadUntil then
                    setReloading(false)
                end
            end))

            F.cleanupAutoReload = function()
                if GunBase[fireName] == wrappedFire then GunBase[fireName] = originalFire end
                if GunBase[reloadName] == wrappedReload then GunBase[reloadName] = originalReload end
                setReloading(false)
            end
        end
    end

    F.isWeaponReloading = F.isWeaponReloading or function()
        return Settings.Reload.IsReloading
    end
end

do
    local gunCache, meleeCache = {}, {}
    local built, scanning = false, false

    local function buildCaches()
        if scanning then return end
        scanning = true
        task.spawn(function()
            gunCache = {}
            meleeCache = {}
            local ok, objs = pcall(function() return getgc(true) end)
            if ok and type(objs) == "table" then
                for _, v in pairs(objs) do
                    if type(v) == "table" then
                        if rawget(v, "FireDelay") ~= nil and rawget(v, "FiringOnCooldown") ~= nil and rawget(v, "OwnerPlayer") ~= nil then
                            gunCache[#gunCache + 1] = v
                        end
                        if rawget(v, "UseDelay") ~= nil and rawget(v, "Range") ~= nil and rawget(v, "UsingOnCooldown") ~= nil and rawget(v, "Destroyed") ~= true then
                            meleeCache[#meleeCache + 1] = v
                        end
                    end
                end
            end
            built = true
            scanning = false
            if F.applyGunMods then pcall(F.applyGunMods) end
        end)
    end

    F.applyGunMods = function()
        if not built then
            buildCaches()
            return
        end
        if Settings.Guns.RapidFire or Settings.Guns.FullAuto then
            local spd = math.max(Settings.Guns.RapidSpeed or 0.02, 0.025)
            for _, v in ipairs(gunCache) do
                pcall(function()
                    if Settings.Guns.RapidFire then
                        rawset(v, "FireDelay", spd)
                        if rawget(v, "FireRate") ~= nil then rawset(v, "FireRate", mFloor(1 / spd)) end
                        rawset(v, "FiringOnCooldown", false)
                        if rawget(v, "OnCooldown") ~= nil then rawset(v, "OnCooldown", false) end
                        if rawget(v, "FireOnCooldown") ~= nil then rawset(v, "FireOnCooldown", false) end
                    end
                    if Settings.Guns.FullAuto then
                        if rawget(v, "FiringType") ~= nil then rawset(v, "FiringType", 2) end
                        if rawget(v, "FireMode") ~= nil then rawset(v, "FireMode", "Auto") end
                        if rawget(v, "Auto") ~= nil then rawset(v, "Auto", true) end
                    end
                end)
            end
        end
        if Settings.Guns.MeleeMods then
            local spd = Settings.Guns.MeleeSpeed or 0.05
            local rng = Settings.Guns.MeleeRange or 5
            for _, v in ipairs(meleeCache) do
                pcall(function()
                    if rawget(v, "Destroyed") == true then return end
                    rawset(v, "UseDelay", spd)
                    rawset(v, "UseTime", spd)
                    rawset(v, "Range", rng)
                    rawset(v, "CharacterRange", rng)
                    rawset(v, "UsingOnCooldown", false)
                end)
            end
        end
    end

    if F.char then buildCaches() end
    F.trackConn(localPlayer.CharacterAdded:Connect(function(char)
        built = false
        buildCaches()
        char.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then
                task.wait(0.1)
                buildCaches()
            end
        end)
    end))
end

do
    local recoilMod = nil
    local applied = false

    local function applyNoRecoil()
        if applied then return end
        if not recoilMod then
            pcall(function()
                recoilMod = require(RS:WaitForChild("Gun"):WaitForChild("Scripts"):WaitForChild("RecoilHandler"))
            end)
        end
        if recoilMod then
            if not recoilMod._o_nS then recoilMod._o_nS = recoilMod.nextStep end
            if not recoilMod._o_sRM then recoilMod._o_sRM = recoilMod.setRecoilMultiplier end
            recoilMod.nextStep = function() end
            recoilMod.setRecoilMultiplier = function() end
            applied = true
        end
    end

    local function revertNoRecoil()
        if not applied then return end
        if recoilMod then
            if recoilMod._o_nS then recoilMod.nextStep = recoilMod._o_nS; recoilMod._o_nS = nil end
            if recoilMod._o_sRM then recoilMod.setRecoilMultiplier = recoilMod._o_sRM; recoilMod._o_sRM = nil end
        end
        applied = false
    end

    F.setNoRecoil = function(on)
        Settings.Guns.NoRecoil = on
        if on then applyNoRecoil() else revertNoRecoil() end
    end

do
    local noSpreadApplied = false
    local originals = {}

    local function applyNoSpread()
        local GunClient, UtilsModule
        pcall(function() GunClient = require(RS.Gun.Scripts.GunClient) end)
        pcall(function() UtilsModule = require(RS.Gun.Scripts.Utils) end)
        if GunClient then
            if GunClient.getBulletSpread and not originals.getBulletSpread then
                originals.getBulletSpread = GunClient.getBulletSpread
                GunClient.getBulletSpread = function() return 0 end
            end
            if GunClient.updateSpreadMult and not originals.updateSpreadMult then
                originals.updateSpreadMult = GunClient.updateSpreadMult
                GunClient.updateSpreadMult = function() end
            end
        end
        if UtilsModule then
            local target = UtilsModule.randomUtils or UtilsModule
            if target.applySpreadToDirection and not originals.applySpreadToDirection then
                originals.applySpreadToDirection = target.applySpreadToDirection
                target.applySpreadToDirection = function(d) return d end
            end
            if target.applyGaussianSpreadToDirection and not originals.applyGaussianSpreadToDirection then
                originals.applyGaussianSpreadToDirection = target.applyGaussianSpreadToDirection
                target.applyGaussianSpreadToDirection = function(d) return d end
            end
        end
        noSpreadApplied = true
    end

    local function revertNoSpread()
        if not noSpreadApplied then return end
        local GunClient, UtilsModule
        pcall(function() GunClient = require(RS.Gun.Scripts.GunClient) end)
        pcall(function() UtilsModule = require(RS.Gun.Scripts.Utils) end)
        if GunClient then
            if originals.getBulletSpread then GunClient.getBulletSpread = originals.getBulletSpread end
            if originals.updateSpreadMult then GunClient.updateSpreadMult = originals.updateSpreadMult end
        end
        if UtilsModule then
            local target = UtilsModule.randomUtils or UtilsModule
            if originals.applySpreadToDirection then target.applySpreadToDirection = originals.applySpreadToDirection end
            if originals.applyGaussianSpreadToDirection then target.applyGaussianSpreadToDirection = originals.applyGaussianSpreadToDirection end
        end
        noSpreadApplied = false
    end

    F.setNoSpread = function(on)
        Settings.NoSpread.Enabled = on
        if on then applyNoSpread() else revertNoSpread() end
    end

do
    local enlarged = {}
    local charConns = {}
    local watchConn = nil
    local watchInit = false

    local function getCharsFolder()
        return Workspace:FindFirstChild("Characters") or (Workspace:FindFirstChild("Spawned") and Workspace.Spawned:FindFirstChild("Characters"))
    end

    local function enlarge(obj)
        if not obj or not obj.Parent then return end
        if enlarged[obj] then return end
        enlarged[obj] = { Size = obj.Size, Color = obj.Color, Transparency = obj.Transparency, Material = obj.Material }
        obj.Size = obj.Size * Settings.Hitbox.Size
        obj.Color = Settings.Hitbox.Color
        obj.Transparency = Settings.Hitbox.Transparency
        obj.Material = Enum.Material.Neon
    end

    local function restore(obj)
        local orig = enlarged[obj]
        if not orig then return end
        if not obj or not obj.Parent then enlarged[obj] = nil return end
        obj.Size = orig.Size
        obj.Color = orig.Color
        obj.Transparency = orig.Transparency
        obj.Material = orig.Material
        enlarged[obj] = nil
    end

    function F.RestoreAllHitboxes()
        for obj in pairs(enlarged) do restore(obj) end
    end

    function F.UpdateAllHitboxes()
        for obj in pairs(enlarged) do
            if obj and obj.Parent then
                obj.Size = enlarged[obj].Size * Settings.Hitbox.Size
                obj.Color = Settings.Hitbox.Color
                obj.Transparency = Settings.Hitbox.Transparency
            end
        end
    end

    function F.ApplyHitboxVisuals()
        for obj in pairs(enlarged) do
            if obj and obj.Parent then
                obj.Transparency = Settings.Hitbox.Transparency
                obj.Color = Settings.Hitbox.Color
            end
        end
    end

    local function processChar(character)
        if not character or charConns[character] then return end
        if Players:GetPlayerFromCharacter(character) == localPlayer then return end

        local connections = {}
        charConns[character] = connections
        local hitbox = character:FindFirstChild("HeadHitbox")
        local humanoid = character:FindFirstChildWhichIsA("Humanoid")

        local function update()
            if not humanoid or not humanoid.Parent then
                humanoid = character:FindFirstChildWhichIsA("Humanoid")
            end
            if not hitbox or not hitbox.Parent then
                hitbox = character:FindFirstChild("HeadHitbox")
            end
            if hitbox and humanoid then
                if Settings.Hitbox.Enabled and humanoid.Health > 0 then enlarge(hitbox) else restore(hitbox) end
            end
        end

        update()
        connections[#connections + 1] = character.ChildAdded:Connect(function(child)
            if child.Name == "HeadHitbox" or child:IsA("Humanoid") then
                task.defer(update)
            end
        end)
        connections[#connections + 1] = character.ChildRemoved:Connect(function(child)
            if child == hitbox then restore(hitbox); hitbox = nil end
        end)
        if humanoid then
            connections[#connections + 1] = humanoid.HealthChanged:Connect(update)
            connections[#connections + 1] = humanoid.Died:Connect(function() if hitbox then restore(hitbox) end end)
        end
        connections[#connections + 1] = character.AncestryChanged:Connect(function()
            if not character.Parent then
                if hitbox then restore(hitbox) end
                local list = charConns[character]
                if list then for _, connection in ipairs(list) do pcall(function() connection:Disconnect() end) end end
                charConns[character] = nil
            end
        end)
    end

    function F.InitHitboxWatchers()
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= localPlayer and player.Character then processChar(player.Character) end
        end

        local chars = getCharsFolder()
        if chars then
            for _, character in ipairs(chars:GetChildren()) do processChar(character) end
            if not watchInit then
                watchInit = true
                watchConn = chars.ChildAdded:Connect(processChar)
            end
        end
    end

    F.cleanupHitboxes = function()
        if watchConn then pcall(function() watchConn:Disconnect() end) watchConn = nil end
        watchInit = false
        for _, list in pairs(charConns) do
            for _, c in ipairs(list) do pcall(function() c:Disconnect() end) end
        end
        charConns = {}
    end
end

    local function toolIcon(tool)
        local icon = ""
        pcall(function() icon = tool.TextureId or "" end)
        if icon ~= "" then return icon end
        for _, key in ipairs({ "Icon", "Image", "TextureId", "ItemIcon" }) do
            local value = tool:GetAttribute(key)
            if type(value) == "string" and value ~= "" then return value end
            local child = tool:FindFirstChild(key)
            if child and child:IsA("StringValue") and child.Value ~= "" then return child.Value end
        end
        return ""
    end

    local function toolAmount(tool)
        for _, key in ipairs({ "Amount", "Count", "Quantity", "Stack", "StackSize" }) do
            local value = tool:GetAttribute(key)
            if type(value) == "number" then return math.max(1, mFloor(value)) end
            local child = tool:FindFirstChild(key)
            if child and (child:IsA("IntValue") or child:IsA("NumberValue")) then
                return math.max(1, mFloor(child.Value))
            end
        end
        return 1
    end

    local function collectTools(plr)
        local aggregated = {}
        local function add(tool, equipped)
            local image = toolIcon(tool)
            local key = sLower(tool.Name) .. "|" .. image .. "|" .. tostring(equipped)
            local existing = aggregated[key]
            if existing then
                existing.Amount = existing.Amount + toolAmount(tool)
            else
                aggregated[key] = {
                    Name = tool.Name, Image = image, Amount = toolAmount(tool), Equipped = equipped
                }
            end
        end

        local char = plr.Character
        if char then
            for _, item in ipairs(char:GetChildren()) do
                if item:IsA("Tool") then add(item, true) end
            end
        end
        local backpack = plr:FindFirstChild("Backpack")
        if backpack then
            for _, item in ipairs(backpack:GetChildren()) do
                if item:IsA("Tool") then add(item, false) end
            end
        end

        local result = {}
        for _, item in pairs(aggregated) do result[#result + 1] = item end
        table.sort(result, function(a, b)
            if a.Equipped ~= b.Equipped then return a.Equipped end
            return sLower(a.Name) < sLower(b.Name)
        end)
        return result
    end

    function F.copterFlyStep(hum)
        if not hum or not hum.SeatPart then F.disableCopterFly() return end
        local target = hum.SeatPart
        local cam = Workspace.CurrentCamera
        hum.Sit = true
        local moveVec = v3(0, 0, 0)
        local camCF = cam.CFrame
        if UIS:IsKeyDown(Enum.KeyCode.W) then moveVec = moveVec + camCF.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec - camCF.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.D) then moveVec = moveVec + camCF.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.A) then moveVec = moveVec - camCF.RightVector end
        local vert = 0
        if UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.RightShift) then vert = 1 end
        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.RightControl) then vert = -1 end
        if Settings.CopterFly.bodyVel and Settings.CopterFly.bodyVel.Parent == target then
            if moveVec.Magnitude > 0 or vert ~= 0 then
                Settings.CopterFly.bodyVel.Velocity = (moveVec + v3(0, vert, 0)).Unit * Settings.CopterFly.Speed
            else
                Settings.CopterFly.bodyVel.Velocity = v3(0, 0, 0)
            end
        end
        if Settings.CopterFly.bodyGyro and Settings.CopterFly.bodyGyro.Parent == target then
            Settings.CopterFly.bodyGyro.CFrame = camCF
        end
    end

                local col = m.isKill and Settings.HM.KillColor or Settings.HM.Color

                    local d = DIRS[j]

do
    local skinConfigs = {
        ["Tung Tung Sahur"] = { assetId = 138151705692565, scale = 1.6, yOffset = 0 },
        ["Minecraft Villager"] = { assetId = 90405905565781, scale = 1.4, yOffset = 0 },
    }
    F.SkinNames = { "Tung Tung Sahur", "Minecraft Villager" }

    local currentModel = nil
    local currentSkinName = nil
    local templateCache = {}
    local originalVisuals = setmetatable({}, { __mode = "k" })
    local applying = false

    local function skinStatus(text, isError)
        pcall(F.Notify, (isError and "Model error: " or "Model: ") .. text, isError and 5 or 3)
    end

    local function setCharacterVisible(character, visible)
        if not character then return end
        for _, inst in ipairs(character:GetDescendants()) do
            if inst:IsA("BasePart") then
                if not visible then
                    if not originalVisuals[inst] then
                        originalVisuals[inst] = {
                            Transparency = inst.Transparency,
                            LocalTransparencyModifier = inst.LocalTransparencyModifier,
                        }
                    end
                    inst.Transparency = 1
                    inst.LocalTransparencyModifier = 1
                else
                    local original = originalVisuals[inst]
                    if original then
                        inst.Transparency = original.Transparency
                        inst.LocalTransparencyModifier = original.LocalTransparencyModifier
                        originalVisuals[inst] = nil
                    end
                end
            elseif inst:IsA("Decal") or inst:IsA("Texture") then
                if not visible then
                    if not originalVisuals[inst] then originalVisuals[inst] = { Transparency = inst.Transparency } end
                    inst.Transparency = 1
                else
                    local original = originalVisuals[inst]
                    if original then inst.Transparency = original.Transparency; originalVisuals[inst] = nil end
                end
            end
        end
    end

    local function normalizeModel(raw)
        if not raw then return nil end
        if raw:IsA("Model") then return raw end
        local nested = raw:FindFirstChildWhichIsA("Model", true)
        if nested then
            nested.Parent = nil
            raw:Destroy()
            return nested
        end
        if raw:IsA("Tool") or raw:IsA("Accessory") then
            local handle = raw:FindFirstChild("Handle")
            if handle then
                local wrapper = Instance.new("Model")
                handle.Parent = wrapper
                wrapper.PrimaryPart = handle
                raw:Destroy()
                return wrapper
            end
        end
        if raw:IsA("BasePart") then
            local wrapper = Instance.new("Model")
            raw.Parent = wrapper
            wrapper.PrimaryPart = raw
            return wrapper
        end
        local anyPart = raw:FindFirstChildWhichIsA("BasePart", true)
        if anyPart then
            local wrapper = Instance.new("Model")
            raw.Parent = wrapper
            return wrapper
        end
        return nil
    end

    local function loadTemplate(name, assetId)
        if templateCache[name] then return templateCache[name] end
        local objects, loadError
        local ok, result = pcall(function()
            return game:GetObjects("rbxassetid://" .. tostring(assetId))
        end)
        if ok and result and #result > 0 then
            objects = result
        else
            loadError = result
            local okInsert, inserted = pcall(function() return InsertService:LoadAsset(assetId) end)
            if okInsert and inserted then objects = { inserted } else loadError = inserted end
        end
        if not objects then return nil, tostring(loadError) end

        local modelTemplate = normalizeModel(objects[1])
        if not modelTemplate then return nil, "asset has no model or BasePart" end
        for _, inst in ipairs(modelTemplate:GetDescendants()) do
            if inst:IsA("Humanoid") or inst:IsA("Animator") or inst:IsA("AnimationController")
                or inst:IsA("Script") or inst:IsA("LocalScript") or inst:IsA("BodyMover") then
                inst:Destroy()
            elseif inst:IsA("BasePart") then
                inst.CanCollide = false
                inst.CanQuery = false
                inst.CanTouch = false
                inst.Anchored = true
                inst.Massless = true
            end
        end
        if not modelTemplate.PrimaryPart then
            modelTemplate.PrimaryPart = modelTemplate:FindFirstChild("HumanoidRootPart", true)
                or modelTemplate:FindFirstChild("Torso", true)
                or modelTemplate:FindFirstChild("UpperTorso", true)
                or modelTemplate:FindFirstChild("Root", true)
                or modelTemplate:FindFirstChildWhichIsA("BasePart", true)
        end
        if not modelTemplate.PrimaryPart then
            modelTemplate:Destroy()
            return nil, "asset has no BasePart"
        end
        modelTemplate.Parent = nil
        templateCache[name] = modelTemplate
        return modelTemplate
    end

    function F.clearSkin(keepSelection)
        if currentModel then pcall(function() currentModel:Destroy() end) currentModel = nil end
        setCharacterVisible(F.char or localPlayer.Character, true)
        if not keepSelection then currentSkinName = nil end
    end

    function F.applySkin(name)
        if applying then return end
        if not Settings.Model.Enabled then return end
        local config = skinConfigs[name]
        if not config then skinStatus("unknown skin " .. tostring(name), true) return end
        local character = F.char or localPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not character or not root then skinStatus("character is not ready", true) return end

        applying = true
        skinStatus("loading " .. name, false)
        local template, err = loadTemplate(name, config.assetId)
        if not template then applying = false; skinStatus(name .. ": " .. tostring(err), true) return end

        F.clearSkin(true)
        config.scale = Settings.Model.Scale or config.scale or 1
        config.yOffset = Settings.Model.YOffset or config.yOffset or 0

        local clone = template:Clone()
        clone.Name = "FloppaHub_Skin_" .. name
        pcall(function() clone:ScaleTo(config.scale or 1) end)
        clone.Parent = Workspace
        pcall(function()
            clone:PivotTo(root.CFrame * cNew(0, config.yOffset or 0, 0))
        end)
        setCharacterVisible(character, false)
        currentModel = clone
        currentSkinName = name
        Settings.Model.Selected = name
        applying = false
        skinStatus(name .. " applied", false)
    end

    function F.setSkinScale(value)
        Settings.Model.Scale = value
        if currentSkinName and skinConfigs[currentSkinName] then
            skinConfigs[currentSkinName].scale = value
        end
        if currentModel then pcall(function() currentModel:ScaleTo(value) end) end
    end

    function F.setSkinYOffset(value)
        Settings.Model.YOffset = value
        if currentSkinName and skinConfigs[currentSkinName] then
            skinConfigs[currentSkinName].yOffset = value
        end
    end

    function F.updateSkinModel()
        if not Settings.Model.Enabled then
            if currentModel then F.clearSkin(true) end
            return
        end
        if not currentModel then return end
        local character = F.char or localPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        local config = currentSkinName and skinConfigs[currentSkinName]
        if not root or not config or not currentModel.Parent then
            F.clearSkin(true)
            return
        end
        pcall(function()
            currentModel:PivotTo(root.CFrame * cNew(0, config.yOffset or 0, 0))
        end)
    end

    function F.cleanupModelChanger()
        F.clearSkin(false)
        for _, template in pairs(templateCache) do pcall(function() template:Destroy() end) end
        table.clear(templateCache)
    end

    F.trackConn(localPlayer.CharacterAdded:Connect(function()

        end) end
    end))

do
    function F.updateHeadTilt()
        local previous = F._headTiltNeck
        if not Settings.AntiAim.HeadTilt then
            if previous and previous.Parent then pcall(function() previous.Transform = cNew() end) end
            F._headTiltNeck = nil
            return
        end

        local ch = F.char
        local torso = ch and (ch:FindFirstChild("UpperTorso") or ch:FindFirstChild("Torso"))
        local neck = torso and torso:FindFirstChild("Neck")
        if not neck or not neck:IsA("Motor6D") then return end
        F._headTiltNeck = neck
        neck.Transform = CFrame.Angles(mRad(Settings.AntiAim.HeadPitch or -45), 0, 0)
    end

    function F.updateAntiAim(dt)
        if not Settings.AntiAim.Enabled then return end
        local char = F.char
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum = F.hum
        if not root or not hum or hum.Health <= 0 then return end

        local mode = Settings.AntiAim.Mode
        local yaw = 0
        if mode == "Spin" then
            Settings.AntiAim.T = (Settings.AntiAim.T + Settings.AntiAim.Speed * dt) % 360
            yaw = Settings.AntiAim.T
        elseif mode == "Jitter" then
            Settings.AntiAim.T = Settings.AntiAim.T + dt
            local flip = mFloor(Settings.AntiAim.T * 15) % 2
            yaw = flip == 0 and Settings.AntiAim.JitterRange or -Settings.AntiAim.JitterRange
        end
        hum.AutoRotate = false
        root.CFrame = cNew(root.Position) * cAngles(0, mRad(yaw), 0)
    end

    function F.updateSpin(dt)
        if not Settings.Spin.Enabled then return end
        local char = F.char
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum = F.hum
        if not root or not hum or hum.Health <= 0 then return end
        hum.AutoRotate = false
        root.CFrame = root.CFrame * cAngles(0, mRad(Settings.Spin.Speed * dt), 0)
    end
end

do
    local lastHit = {}
    local recentDamage = setmetatable({}, { __mode = "k" })
    local dmgConns = {}
    local deathConns = {}
    local humHealth = {}

    local function isOurDamage(plr, char)
        if F.lastConfirmedShotPlayer == plr and (osClock() - (F.lastConfirmedShotTime or 0)) <= 6 then
            return true
        end
        if F.lastTargetPlayer == plr and (osClock() - (F.lastTargetTime or 0)) <= 1.25 then
            return true
        end
        local part = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
        return part ~= nil and F.isOnOurShot(part.Position)
    end

    local deathSeen = setmetatable({}, { __mode = "k" })
    local function onPlayerDied(plr, char)
        local now = osClock()
        if deathSeen[plr] and now - deathSeen[plr] <= 1.5 then return end
        deathSeen[plr] = now
        if lastHit[plr] and (now - lastHit[plr]) <= 6 then
            F.playKillSound()
            local root = char and char:FindFirstChild("HumanoidRootPart")
            local pos = root and root.Position
            if Settings.HM.Enabled and pos then
                F.spawnHitMarker(pos, true)
            end
        end
        lastHit[plr] = nil
    end

    local function onDamaged(plr, char, dmg, newHealth)
        if F.Unloaded or dmg <= 0.01 then return end
        if not isOurDamage(plr, char) then return end

        local now = osClock()
        local previous = recentDamage[plr]
        if previous and now - previous.time <= 0.2
            and mAbs(previous.health - newHealth) <= 0.001
            and mAbs(previous.damage - dmg) <= 0.001 then
            return
        end
        recentDamage[plr] = { time = now, health = newHealth, damage = dmg }

        local part = char:FindFirstChild(Settings.SA.TargetPart) or char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
        if not part then return end
        lastHit[plr] = osClock()

        if Settings.HM.Enabled then F.spawnHitMarker(part.Position, false) end
        if Settings.HL.Enabled then
            F.showHitLog(plr.DisplayName, part.Name, sFormat("%.0f", dmg), sFormat("%.0f", newHealth), newHealth <= 0)
        end
    end

    local function trackDeath(plr)
        if plr == localPlayer then return end
        local function hc(c)
            task.spawn(function()
                local h = c:WaitForChild("Humanoid", 10)
                if h then
                    local oldDeath = deathConns[h]
                    if oldDeath then pcall(function() oldDeath:Disconnect() end) end
                    deathConns[h] = h.Died:Connect(function()
                        if Settings.KillNotify then
                            pcall(F.Notify, plr.DisplayName .. " (" .. plr.Name .. ") died!", 3)
                        end
                        onPlayerDied(plr, c)
                    end)
                end
            end)
        end
        if plr.Character then task.spawn(hc, plr.Character) end
        F.trackConn(plr.CharacterAdded:Connect(hc))
    end
    for _, p in ipairs(Players:GetPlayers()) do trackDeath(p) end
    F.trackConn(Players.PlayerAdded:Connect(trackDeath))

    local function trackDamage(plr)
        if plr == localPlayer then return end
        local function hc(char)
            task.spawn(function()
                local hum = char:WaitForChild("Humanoid", 10)
                if not hum then return end
                local old = dmgConns[hum]
                if old then pcall(function() old:Disconnect() end) end
                humHealth[hum] = hum.Health
                dmgConns[hum] = hum.HealthChanged:Connect(function(new)
                    local prev = humHealth[hum] or new
                    humHealth[hum] = new
                    if new < prev - 0.05 then
                        onDamaged(plr, char, prev - new, new)
                    end
                end)
            end)
        end
        if plr.Character then hc(plr.Character) end
        F.trackConn(plr.CharacterAdded:Connect(hc))
    end
    for _, plr in ipairs(Players:GetPlayers()) do trackDamage(plr) end
    F.trackConn(Players.PlayerAdded:Connect(trackDamage))

    F.cleanupDamageTracking = function()
        for humanoid, connection in pairs(dmgConns) do
            pcall(function() connection:Disconnect() end)
            dmgConns[humanoid] = nil
        end
        for humanoid, connection in pairs(deathConns) do
            pcall(function() connection:Disconnect() end)
            deathConns[humanoid] = nil
        end
        table.clear(humHealth)
        table.clear(lastHit)
        table.clear(recentDamage)
    end
end

        statsLbl.TextColor3 = rgb(160, 160, 180)
        statsLbl.TextXAlignment = Enum.TextXAlignment.Right
        statsLbl.Text = ""
        statsLbl.Parent = wm

        task.spawn(function()
            local ok, img = pcall(function()
                return Players:GetUserThumbnailAsync(
                    localPlayer.UserId,
                    Enum.ThumbnailType.HeadShot,
                    Enum.ThumbnailSize.Size100x100
                )
            end)
            if ok and img then
                avatar.Image = img
            end
        end)

        local fpsAcc = 0
        local fpsCount = 0
        local wmConn
        local lastUpdate = 0
        wmConn = RunService.RenderStepped:Connect(function(dt)
            if F.Unloaded then
                if wmConn then wmConn:Disconnect() end
                return
            end
            if not Settings.Watermark.Enabled then
                wm.Visible = false
                return
            end
            wm.Visible = true
            fpsAcc = fpsAcc + dt
            fpsCount = fpsCount + 1
            local now = tick()
            if now - lastUpdate < 0.25 then return end
            lastUpdate = now

            local fps = fpsCount > 0 and mFloor(fpsCount / fpsAcc) or 0
            fpsAcc = 0
            fpsCount = 0

            local ping = 0
            pcall(function()
                ping = mFloor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
            end)

            local t = os.date("%H:%M:%S")

            rightLbl.Text = string.format("%s  |  %d ms", t, ping)
            statsLbl.Text = string.format("FPS: %d  |  %s", fps, localPlayer.DisplayName)
        end)

        F.watermarkGui = wmGui
    end

    createWatermark()
end

F.Notify = function(msg, dur)
    pcall(function() Library:Notify({ Title = 'FLOPPA HUB', Content = tostring(msg), Duration = dur or 3 }) end)
end

do
    local CFG_DIR = 'FloppaHub/configs'
    local CFG_SKIP = { MenuKeybind = true }
    pcall(function()
        if not isfolder('FloppaHub') then makefolder('FloppaHub') end
        if not isfolder(CFG_DIR) then makefolder(CFG_DIR) end
    end)

    local function cfgPath(name) return CFG_DIR .. '/' .. name .. '.json' end

    local function cfgNames()
        local out = {}
        local ok, files = pcall(listfiles, CFG_DIR)
        if ok and files then
            for _, f in ipairs(files) do
                local name = tostring(f):match('([^/\\]+)%.json$')
                if name then table.insert(out, name) end
            end
        end
        table.sort(out)
        return out
    end

    local function cfgSave(name)
        local data = { toggles = {}, options = {} }
        for id, t in pairs(Toggles) do
            if not CFG_SKIP[id] then data.toggles[id] = t.Value end
        end
        for id, o in pairs(Options) do
            if not CFG_SKIP[id] then
                if o.Type == 'ColorPicker' then
                    data.options[id] = { kind = 'color', value = o.Value:ToHex(), t = o.Transparency or 0 }
                elseif o.Type == 'KeyPicker' then
                    data.options[id] = { kind = 'key', value = { o.Value, o.Mode } }
                else
                    data.options[id] = { kind = 'val', value = o.Value }
                end
            end
        end
        writefile(cfgPath(name), HttpService:JSONEncode(data))
    end

    local function cfgLoad(name)
        local data = HttpService:JSONDecode(readfile(cfgPath(name)))
        for id, v in pairs(data.toggles or {}) do
            local t = Toggles[id]
            if t and not CFG_SKIP[id] then pcall(function() t:SetValue(v) end) end
        end
        for id, e in pairs(data.options or {}) do
            local o = Options[id]
            if o and not CFG_SKIP[id] then
                pcall(function()
                    if e.kind == 'color' then
                        o:SetValueRGB(Color3.fromHex(e.value), e.t)
                    else
                        o:SetValue(e.value)
                    end
                end)
            end
        end
    end

    local CFGBox = Tabs.Misc:AddLeftGroupbox('Configs')
    CFGBox:AddInput('cfg_Name', { Text = 'Config name', Default = '', Placeholder = 'name' })
    local cfgList = CFGBox:AddDropdown('cfg_List', { Text = 'Saved configs', Values = cfgNames(), AllowNull = true })

    local function refreshConfigs()
        cfgList:SetValues(cfgNames())
        cfgList:SetValue(nil)
    end

    CFGBox:AddButton({
        Text = 'Save',
        Func = function()
            local name = Options.cfg_Name.Value:match('^%s*(.-)%s*$')
            if name == '' then name = cfgList.Value or '' end
            if not name:match('^[%w_%-]+$') then
                Library:Notify('invalid config name', 3)
                return
            end
            local ok, err = pcall(cfgSave, name)
            Library:Notify(ok and ('saved ' .. name) or ('save failed: ' .. tostring(err)), 2)
            refreshConfigs()
        end
    })

    CFGBox:AddButton({
        Text = 'Load',
        Func = function()
            local name = cfgList.Value
            if not name then
                Library:Notify('select a config first', 2)
                return
            end
            local ok, err = pcall(cfgLoad, name)
            Library:Notify(ok and ('loaded ' .. name) or ('load failed: ' .. tostring(err)), 2)
        end
    })

    CFGBox:AddButton({ Text = 'Refresh list', Func = refreshConfigs })
    CFGBox:AddButton({
        Text = 'Delete (double click)',
        DoubleClick = true,
        Func = function()
            local name = cfgList.Value
            if not name then return end
            pcall(delfile, cfgPath(name))
            Library:Notify('deleted ' .. name, 2)
            refreshConfigs()
        end
    })
end

    F.Unloaded = true
    pcall(function() RunService:UnbindFromRenderStep('FloppaHub_ESP') end)
    for player, _ in pairs(EspPlayers) do
        pcall(function() RemoveESP(player) end)
    end
    pcall(function() ScreenGui:Destroy() end)
    pcall(function() F.fovGui:Destroy() end)
    pcall(function() if F.watermarkGui then F.watermarkGui:Destroy() end end)
    pcall(F.cleanupHitLogs)
    pcall(F.cleanupDamageTracking)
    pcall(F.cleanupHitMarkers)
    pcall(F.cleanupHitboxes)
    pcall(F.cleanupWorldConns)
    pcall(F.cleanupInventoryInspector)
    pcall(F.cleanupNoFall)
    pcall(F.cleanupHSChar)
    pcall(F.cleanupFB)
    pcall(F.cleanupSky)
    pcall(F.cleanupReload)
    pcall(F.cleanupAutoReload)
    pcall(F.cleanupModelChanger)
    pcall(F.cleanupBacktrack)
    pcall(F.destroyAllTracers)
    pcall(F.restoreHS)
    pcall(F.RestoreAllHitboxes)
    pcall(function()
        local ch = localPlayer.Character
        if ch then
            local hrp = ch:FindFirstChild("HumanoidRootPart")
            if hrp then
                local bv = hrp:FindFirstChild("WSV")
                if bv then bv:Destroy() end
            end
        end
    end)
    pcall(F.disableCopterFly)
    for _, c in pairs(F.conns) do pcall(function() c:Disconnect() end) end
    F.conns = {}
end)
